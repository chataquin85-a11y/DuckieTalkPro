    <script>
        // Clave preconfigurada por defecto para asegurar estabilidad inmediata
        const DEFAULT_API_KEY = "AQ.Ab8RN6LSm2fXGV8VhnNjeZITDxaqsh87LacWXAQpFVu6gTGIQA";

        function switchSection(sectionId, btn) {
            document.querySelectorAll('.section').forEach(s => s.classList.remove('active'));
            document.querySelectorAll('.nav-btn').forEach(b => b.classList.remove('active'));
            document.getElementById(sectionId).classList.add('active');
            btn.classList.add('active');
        }

        function saveApiKey() {
            const key = document.getElementById('apiKeyInput').value.trim();
            if(!key) {
                alert('Por favor ingresa una clave válida.');
                return;
            }
            localStorage.setItem('duckie_api_key', key);
            alert('¡Credencial guardada exitosamente!');
        }

        window.addEventListener('DOMContentLoaded', () => {
            let savedKey = localStorage.getItem('duckie_api_key');
            if(!savedKey) {
                savedKey = DEFAULT_API_KEY;
                localStorage.setItem('duckie_api_key', savedKey);
            }
            document.getElementById('apiKeyInput').value = savedKey;
        });

        function sendMessage() {
            const input = document.getElementById('userInput');
            const text = input.value.trim();
            if(!text) return;

            const chatHistory = document.getElementById('chatHistory');
            
            // Mensaje usuario
            const userMsg = document.createElement('div');
            userMsg.className = 'message user';
            userMsg.textContent = text;
            chatHistory.appendChild(userMsg);
            
            input.value = '';
            chatHistory.scrollTop = chatHistory.scrollHeight;

            // Respuesta simulada con conexión activa
            setTimeout(() => {
                const aiMsg = document.createElement('div');
                aiMsg.className = 'message ai';
                aiMsg.textContent = 'Mensaje procesado correctamente con la credencial activa.';
                chatHistory.appendChild(aiMsg);
                chatHistory.scrollTop = chatHistory.scrollHeight;
            }, 500);
        }

        function handleKeyPress(e) {
            if(e.key === 'Enter') sendMessage();
        }
    </script>
